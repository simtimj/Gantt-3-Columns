/* Gantt PWA service worker (2026-11-21). Built by tools/build-pwa.mjs; BUILD changes with every page change.
   The app shell opens from cache (fast, works offline). Server calls go to script.google.com and are never
   cached. A new version waits until the page's "↻ Update ready" button says go. */
var BUILD = '__BUILD__';
var CACHE = 'gantt-' + BUILD;
var SHELL = __SHELL__;

self.addEventListener('install', function (e) {
  e.waitUntil(caches.open(CACHE).then(function (c) { return c.addAll(SHELL.map(function (u) { return new Request(u, { cache: 'reload' }); })); }));
});

// Old copies are deleted after the new version takes over, not before (2026-11-26, Speed Log: reloads waited ~1 s for
// first byte; the page fetch waits for activation, and deleting a big old cache sat in front of it).
self.addEventListener('activate', function (e) {
  e.waitUntil(self.clients.claim());
  setTimeout(function () {
    caches.keys().then(function (keys) {
      keys.filter(function (k) { return k.indexOf('gantt-') === 0 && k !== CACHE; }).forEach(function (k) { caches.delete(k); });
    });
  }, 3000);
});

self.addEventListener('message', function (e) { if (e.data === 'skipWaiting') self.skipWaiting(); });

// Calendar libraries (2026-11-23, speed step A): FullCalendar / Luxon come from jsdelivr at pinned versions, so a copy
// never goes stale; kept on the device (own cache, survives app updates) so the calendar paints without the network.
var LIBS = 'libs-v1';
function isPinnedLib(url) { return url.hostname === 'cdn.jsdelivr.net' && /\/npm\/[^/]*@\d[^/]*\//.test(url.pathname); }

self.addEventListener('fetch', function (e) {
  var req = e.request, url = new URL(req.url);
  if (req.method === 'GET' && isPinnedLib(url)) {
    e.respondWith(caches.open(LIBS).then(function (c) {
      return c.match(req).then(function (hit) {
        return hit || fetch(req).then(function (res) { if (res && (res.ok || res.type === 'opaque')) c.put(req, res.clone()); return res; });
      });
    }));
    return;
  }
  if (req.method !== 'GET' || url.origin !== self.location.origin) return;
  if (req.mode === 'navigate') {
    var page = /tracker(\.html)?$/.test(url.pathname) ? 'tracker.html' : 'index.html';
    e.respondWith(caches.open(CACHE).then(function (c) {
      return c.match(page).then(function (hit) { return hit || fetch(req); });
    }));
    return;
  }
  e.respondWith(caches.match(req, { ignoreSearch: true }).then(function (hit) { return hit || fetch(req); }));
});
