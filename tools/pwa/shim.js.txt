/* Gantt PWA shim (2026-11-21, docs/NOTES.md H). Injected first in the hosted page by tools/build-pwa.mjs.
   - google.script.run → fetch to the "Gantt API" router (POST text/plain, so no CORS preflight), with the app key.
   - Main page: the last data seen on this device opens at once (if it's for today), then fwSync(true) catches up.
   - Offline: read-only + banner. Service worker updates: "↻ Update ready" button, never an auto-reload.
   - Database step (2026-11-22, NOTES H3): with a Firebase config, Keep In Mind reads/writes Firestore directly
     (Google sign-in, his account only), works offline (Firestore queues the edits) and syncs live.
   - Days (2026-11-22, NOTES H4): Today / Notes / Plan / Thoughts save to a doc per day, newest wins per section,
     editable offline for today ± 7 days. */
(function () {
  'use strict';
  var API = '__API__', PAGE = '__PAGE__', BUILD = '__BUILD__', FB_CFG = __FIREBASE__;
  var KEY_LS = 'pwa.key', DATA_LS = 'pwa.initial.v1';
  var ls = {
    get: function (k) { try { return localStorage.getItem(k); } catch (e) { return null; } },
    set: function (k, v) { try { localStorage.setItem(k, v); return true; } catch (e) { return false; } },
    del: function (k) { try { localStorage.removeItem(k); } catch (e) {} }
  };
  window.__PWA__ = { build: BUILD, page: PAGE };

  // Key from the setup link (#k=…): the fragment never reaches a server; save it, then strip it from the URL.
  var m = location.hash.match(/[#&]k=([^&]+)/);
  if (m) ls.set(KEY_LS, decodeURIComponent(m[1]));
  // Look + layout settings copied from the old app (setup link &s=…): applied before the app's scripts read them.
  var sm = location.hash.match(/[#&]s=([^&]+)/);
  if (sm) {
    try {
      var st = JSON.parse(decodeURIComponent(sm[1])), n = 0;
      Object.keys(st).forEach(function (k) {
        if (/^(fw\.|focusWorkspace|tt\.ui\.)/.test(k) && typeof st[k] === 'string' && ls.set(k, st[k])) n++;
      });
      window.__PWA__.settingsCopied = n;
    } catch (e) {}
  }
  if (m || sm) { try { history.replaceState(null, '', location.pathname + location.search); } catch (e) {} }
  function key() { return ls.get(KEY_LS) || ''; }

  /* ---------- offline / read-only ---------- */
  var offline = !navigator.onLine;
  function setOffline(on) {
    on = !!on;
    if (on === offline && document.body && document.body.classList.contains('pwa-offline') === on) return;
    offline = on;
    if (!document.body) return;
    document.body.classList.toggle('pwa-offline', on);
    var bar = document.getElementById('pwaOffline');
    if (on && !bar) {
      bar = document.createElement('div');
      bar.id = 'pwaOffline';
      bar.textContent = fb.offlineText();
      document.body.appendChild(bar);
    } else if (!on && bar) bar.remove();
  }
  window.addEventListener('online', function () { setOffline(false); if (typeof window.fwSync === 'function') try { window.fwSync(true); } catch (e) {} });
  window.addEventListener('offline', function () { setOffline(true); });
  document.addEventListener('DOMContentLoaded', function () { if (offline) { offline = !offline; setOffline(true); } });
  function block(e) {
    if (!offline) return;
    if (e.target && e.target.closest && e.target.closest('#pwaKeyBox')) return;
    if (fb.editable(e.target)) return;
    e.preventDefault(); e.stopImmediatePropagation();
    flash();
  }
  ['beforeinput', 'paste', 'drop', 'cut'].forEach(function (t) { document.addEventListener(t, block, true); });
  document.addEventListener('click', function (e) {
    if (offline && e.target && e.target.matches && e.target.matches('input[type=checkbox]')) block(e);
  }, true);
  var flashT = 0;
  function flash() {
    var bar = document.getElementById('pwaOffline'); if (!bar) return;
    bar.classList.add('pwa-flash'); clearTimeout(flashT);
    flashT = setTimeout(function () { bar.classList.remove('pwa-flash'); }, 600);
  }

  /* ---------- key prompt ---------- */
  function needKey(msg) {
    if (document.getElementById('pwaKeyBox')) return;
    var show = function () {
      var box = document.createElement('div');
      box.id = 'pwaKeyBox';
      box.innerHTML = '<div><b>App key needed</b><p>' + (msg || 'This device has no key yet.') +
        ' Easiest: open your private /dev link with <code>?app=pwa</code> at the end, then tap “Open the Gantt app”.</p>' +
        '<input id="pwaKeyIn" type="password" autocomplete="off" placeholder="…or paste the key"><button id="pwaKeyOk">Save</button></div>';
      document.body.appendChild(box);
      document.getElementById('pwaKeyOk').onclick = function () {
        var v = document.getElementById('pwaKeyIn').value.trim();
        if (v) { ls.set(KEY_LS, v); location.reload(); }
      };
    };
    if (document.body) show(); else document.addEventListener('DOMContentLoaded', show);
  }
  if (!key()) needKey();

  /* ---------- Firestore: Keep In Mind (database step, 2026-11-22, docs/NOTES.md → H3) ----------
     getKim / saveKim go to Firestore doc gantt/kim = { html, at, base, dev } instead of the server. The SDK loads
     after the page (no cost to the first paint) from this site (cached by the service worker, so it works offline).
     Offline edits are queued by Firestore and sent when back online. Newest edit wins: the rules refuse an update
     whose `at` isn't newer; the refused copy goes to ganttBackup (→ Backup tab via the Sheet mirror).
     Not signed in, no doc yet, or the SDK not ready within 4 s → the old server path (getKim / saveKim). */
  var fb = (function () {
    var off = { handles: function () { return false; }, editable: function () { return false; },
      offlineText: function () { return 'Offline · read-only'; }, overlay: function (n, r) { return r; } };
    if (!FB_CFG || !FB_CFG.apiKey || PAGE !== 'index') return off;
    var AREAS = FB_CFG.areas || ['kim'], KIM = AREAS.indexOf('kim') >= 0, DAYS = AREAS.indexOf('days') >= 0;
    var NAMES = {}, UNCONF = 'pwa.kim.unconfirmed', DEV_LS = 'pwa.device';
    if (KIM) { NAMES.getKim = 1; NAMES.saveKim = 1; }
    var DAY_NAMES = { saveDay: 1, savePlanPad: 1, saveThoughtPad: 1, saveScratchPad: 1, getScratchPad: 1 };
    if (DAYS) Object.keys(DAY_NAMES).forEach(function (n) { NAMES[n] = 1; });
    var REF = AREAS.indexOf('ref') >= 0;
    var REF_NAMES = { saveReferenceItem: 1, addReferenceItem: 1, saveReferenceOrder: 1, saveReferenceExpanded: 1, deleteReferenceItem: 1, getArchive: 1, restoreArchivedItem: 1 };
    if (REF) Object.keys(REF_NAMES).forEach(function (n) { NAMES[n] = 1; });
    var OWNER = String(FB_CFG.owner || '').toLowerCase(), cfg = Object.assign({}, FB_CFG);
    delete cfg.owner; delete cfg._note; delete cfg.areas;
    var db = null, user = null, lastAt = null, watching = false;
    var dev = ls.get(DEV_LS);
    if (!dev) { dev = (/Android|iPhone|iPad/i.test(navigator.userAgent) ? 'phone-' : 'pc-') + Math.random().toString(36).slice(2, 8); ls.set(DEV_LS, dev); }

    function load(src) {
      return new Promise(function (ok, bad) {
        var el = document.createElement('script'); el.src = src; el.async = false;
        el.onload = ok; el.onerror = function () { bad(new Error('Couldn’t load ' + src)); };
        document.head.appendChild(el);
      });
    }
    var boot = new Promise(function (ok) {
      if (document.readyState === 'complete') setTimeout(ok, 200);
      else window.addEventListener('load', function () { setTimeout(ok, 200); });
    }).then(function () {
      return Promise.all(['fb/firebase-app-compat.js', 'fb/firebase-auth-compat.js', 'fb/firebase-firestore-compat.js'].map(load));
    }).then(function () {
      firebase.initializeApp(cfg);
      db = firebase.firestore();
      return db.enablePersistence({ synchronizeTabs: true }).catch(function (e) { console.warn('Firestore offline copy off: ' + (e.code || e)); });
    }).then(function () {
      return new Promise(function (ok) {
        firebase.auth().onAuthStateChanged(function (u) {
          if (u && String(u.email || '').toLowerCase() !== OWNER) { firebase.auth().signOut(); u = null; }
          user = u || null;
          signButton(!user);
          if (user) { if (KIM) watch(); if (DAYS) days.watch(); if (REF) refs.watch(); }
          ok();
        });
      });
    });
    boot.catch(function (e) { console.warn('Firebase: ' + (e && e.message || e)); });

    function withTimeout(p, ms) {
      return Promise.race([p, new Promise(function (_, bad) { setTimeout(function () { bad(new Error('timeout')); }, ms); })]);
    }
    function doc() { return db.collection('gantt').doc('kim'); }
    function shape(snap) {
      var d = snap && snap.exists ? snap.data() : null;
      return d ? { html: String(d.html || ''), h: 'f' + d.at, at: Number(d.at) || 0, fresh: false, build: '' } : null;
    }
    // Refused by the database (rules, sign-in) → throw, so run() falls back to the server (Sheet) path instead
    // of showing an empty list or accepting an edit that can't be saved.
    function refused(e) { return e && /permission-denied|unauthenticated|failed-precondition/.test(e.code || ''); }
    function read() {
      var cacheOnly = function () { return doc().get({ source: 'cache' }).catch(function () { return null; }); };
      return (navigator.onLine ? doc().get({ source: 'server' }).catch(function (e) { if (refused(e)) throw e; return cacheOnly(); }) : cacheOnly()).then(shape);
    }
    function backup(html, reason) {
      db.collection('ganttBackup').add({ area: 'kim', html: String(html || ''), at: Date.now(), reason: reason, dev: dev })
        .catch(function (e) { console.warn('Backup not saved: ' + (e.code || e)); });
    }
    function unconf() { try { return JSON.parse(ls.get(UNCONF) || 'null'); } catch (e) { return null; } }
    function save(html, baseH) {
      return read().then(function (cur) {
        if (cur && cur.h !== baseH) return { conflict: true, html: cur.html, h: cur.h, build: '' };
        html = String(html || '');
        if (cur && cur.html.length > 400 && html.length < cur.html.length * 0.6) backup(cur.html, 'big shrink');
        var at = Math.max(Date.now(), (cur ? cur.at : 0) + 1);
        var rec = { html: html, at: at, base: cur ? cur.at : 0, dev: dev };
        ls.set(UNCONF, JSON.stringify({ html: html, at: at }));
        // Not awaited: offline, this only resolves once the queued edit reaches the server.
        doc().set(rec).then(function () {
          var u = unconf(); if (u && u.at === at) ls.del(UNCONF);
        }, function (e) {
          if (e && e.code === 'permission-denied') {
            backup(html, 'older edit lost (newest wins)');
            var u = unconf(); if (u && u.at === at) ls.del(UNCONF);
            if (typeof window.fwKimReload === 'function') window.fwKimReload();
          } else console.warn('Keep In Mind not saved to the database: ' + (e.code || e));
        });
        lastAt = at;
        return { h: 'f' + at, at: at, build: '' };
      });
    }
    // Live sync: another device saved → the list merges it in. Also settles an edit left over from an earlier
    // session: confirmed → forget it; the server kept a different copy → the leftover goes to Backup.
    function watch() {
      if (watching) return; watching = true;
      doc().onSnapshot({ includeMetadataChanges: true }, function (snap) {
        if (snap.metadata.fromCache || snap.metadata.hasPendingWrites || !snap.exists) return;
        var at = Number(snap.data().at) || 0, u = unconf();
        if (u) { if (u.at !== at) backup(u.html, 'older edit lost (newest wins)'); ls.del(UNCONF); }
        if (lastAt !== null && at !== lastAt && typeof window.fwKimReload === 'function') window.fwKimReload();
        lastAt = at;
      }, function (e) { watching = false; console.warn('Keep In Mind live sync stopped: ' + (e.code || e)); });
    }
    function signButton(show) {
      if (!document.body) { document.addEventListener('DOMContentLoaded', function () { signButton(show); }); return; }
      var b = document.getElementById('pwaSignIn');
      if (show && !b) {
        b = document.createElement('button');
        b.id = 'pwaSignIn'; b.type = 'button'; b.textContent = DAYS ? 'Sign in · sync + offline edits' : 'Sign in · Keep In Mind sync';
        b.onclick = signIn;
        document.body.appendChild(b);
      } else if (!show && b) b.remove();
    }
    function signIn() {
      var p = new firebase.auth.GoogleAuthProvider();
      p.setCustomParameters({ login_hint: OWNER, prompt: 'select_account' });
      firebase.auth().signInWithPopup(p).catch(function (e) {
        if (e && /popup-blocked|operation-not-supported/.test(e.code || '')) return firebase.auth().signInWithRedirect(p);
        if (e && e.code !== 'auth/popup-closed-by-user' && e.code !== 'auth/cancelled-popup-request') alert('Sign-in didn’t work: ' + (e.message || e));
      });
    }
    // cellHash_ (Code.gs): first 8 bytes of the MD5 of the UTF-8 text, as hex. Same versions as the server's.
    function hash(text) {
      var b = new TextEncoder().encode(String(text || '')), n = b.length, words = new Array(((n + 8) >> 6) + 1 << 4).fill(0), i;
      for (i = 0; i < n; i++) words[i >> 2] |= b[i] << (i % 4 * 8);
      words[n >> 2] |= 0x80 << (n % 4 * 8);
      words[words.length - 2] = n * 8;
      var S = [7, 12, 17, 22, 5, 9, 14, 20, 4, 11, 16, 23, 6, 10, 15, 21], K = [];
      for (i = 0; i < 64; i++) K[i] = Math.floor(Math.abs(Math.sin(i + 1)) * 4294967296) | 0;
      var h = [0x67452301, 0xefcdab89 | 0, 0x98badcfe | 0, 0x10325476];
      for (var o = 0; o < words.length; o += 16) {
        var a = h[0], c = h[2], d = h[3], bb = h[1];
        for (i = 0; i < 64; i++) {
          var f, g, r = i >> 4;
          if (r === 0) { f = (bb & c) | (~bb & d); g = i; }
          else if (r === 1) { f = (d & bb) | (~d & c); g = (5 * i + 1) % 16; }
          else if (r === 2) { f = bb ^ c ^ d; g = (3 * i + 5) % 16; }
          else { f = c ^ (bb | ~d); g = (7 * i) % 16; }
          var t = d; d = c; c = bb;
          var x = (a + f + K[i] + words[o + g]) | 0, sh = S[r * 4 + i % 4];
          bb = (bb + ((x << sh) | (x >>> (32 - sh)))) | 0; a = t;
        }
        h[0] = (h[0] + a) | 0; h[1] = (h[1] + bb) | 0; h[2] = (h[2] + c) | 0; h[3] = (h[3] + d) | 0;
      }
      var out = '';
      for (i = 0; i < 8; i++) out += ((h[i >> 2] >>> (i % 4 * 8)) & 255).toString(16).padStart(2, '0');
      return out;
    }
    /* ---------- Days (H4): doc per day ganttDays/{date}, newest wins per section ----------
       Same answers as the server's saveDay / savePlanPad / saveThoughtPad / saveScratchPad / getScratchPad, with
       the same hashes (MD5 below = cellHash_). A section not in the database yet: online → the server (it writes the
       doc); offline → checked against this device's last copy. Unconfirmed edits stay in localStorage and are laid
       over server answers and the opening data until the database has them. */
    function daysArea() {
      var FIELDS = ['tasks', 'journal', 'plan', 'tho'], HKEY = { tasks: 'th', journal: 'jh', plan: 'ph', tho: 'hh' };
      var DU = 'pwa.days.unconfirmed', WIN = 7, seen = {}, watching = false;
      function dayDoc(d) { return db.collection('ganttDays').doc(d); }
      function ymd(t) { var d = new Date(t); return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0'); }
      function win() { var n = Date.now(); return [ymd(n - WIN * 864e5), ymd(n + WIN * 864e5)]; }
      function inWin(d) { var w = win(); return !!d && d >= w[0] && d <= w[1]; }
      function un() { try { return JSON.parse(ls.get(DU) || '{}') || {}; } catch (e) { return {}; } }
      function unSet(m) { if (Object.keys(m).length) ls.set(DU, JSON.stringify(m)); else ls.del(DU); }
      function unPut(d, f, v, at) { var m = un(); m[d + '|' + f] = { v: v, at: at }; unSet(m); }
      function unDrop(d, f, at) { var m = un(), k = d + '|' + f; if (m[k] && (at == null || m[k].at === at)) { delete m[k]; unSet(m); } }
      // This device's last copy of a section (offline check): an unconfirmed edit, else the stored opening data.
      function localCopy(d, f) {
        var u = un()[d + '|' + f]; if (u) return u.v;
        try {
          var c = JSON.parse(ls.get(DATA_LS) || 'null'), day = c && c.data && (c.data.days || []).filter(function (x) { return x.date === d; })[0];
          return day && day[f] != null ? String(day[f]) : null;
        } catch (e) { return null; }
      }
      function readDoc(d) {
        var cache = function () { return dayDoc(d).get({ source: 'cache' }).catch(function () { return null; }); };
        return (navigator.onLine ? dayDoc(d).get({ source: 'server' }).catch(function (e) { if (refused(e)) throw e; return cache(); }) : cache()).then(function (snap) {
          return snap && snap.exists ? snap.data() : null;
        });
      }
      // Current copy of each wanted section, or null when one isn't known here (→ server path).
      function current(d, fields) {
        return readDoc(d).then(function (doc) {
          var cur = { _doc: doc };
          for (var i = 0; i < fields.length; i++) {
            var f = fields[i];
            if (doc && doc[f + 'At']) cur[f] = String(doc[f] == null ? '' : doc[f]);
            else if (navigator.onLine) return null;
            else { var lc = localCopy(d, f); if (lc == null) return null; cur[f] = lc; }
          }
          return cur;
        });
      }
      function backup(d, f, html, reason) {
        db.collection('ganttBackup').add({ area: 'days', date: d, field: f, html: String(html || ''), at: Date.now(), reason: reason, dev: dev })
          .catch(function (e) { console.warn('Backup not saved: ' + (e.code || e)); });
      }
      function openTasks(h) { return (String(h || '').match(/class="check-line(?![^"]*\bdone\b)[^"]*"/g) || []).length; }
      // Write the given sections (not awaited: offline it resolves only once the queued edit reaches the database).
      function write(d, put, cur) {
        var doc = cur._doc || {}, at = Date.now(), rec = { date: d, dev: dev, up: firebase.firestore.FieldValue.serverTimestamp() };
        Object.keys(put).forEach(function (f) { at = Math.max(at, (Number(doc[f + 'At']) || 0) + 1); });
        rec.at = at;
        Object.keys(put).forEach(function (f) {
          var old = cur[f];
          if (old.length > 400 && put[f].length < old.length * 0.6) backup(d, f, old, 'big shrink');
          else if (f === 'tasks' || f === 'journal') { if (openTasks(put[f]) < openTasks(old)) backup(d, f, old, 'open task removed'); }
          rec[f] = put[f]; rec[f + 'At'] = at; rec[f + 'Base'] = hash(old);
          unPut(d, f, put[f], at);
        });
        dayDoc(d).set(rec, { merge: true }).then(function () {
          Object.keys(put).forEach(function (f) { unDrop(d, f, at); });
        }, function (e) {
          if (e && e.code === 'permission-denied') {
            Object.keys(put).forEach(function (f) { backup(d, f, put[f], 'older edit lost (newest wins)'); unDrop(d, f, at); });
            if (typeof window.fwSync === 'function') try { window.fwSync(false); } catch (x) {}
          } else console.warn('Day not saved to the database: ' + (e.code || e));
        });
      }
      // The server's rule for one section (syncCol_ / the pads): same as the copy the app started from → write.
      function col(cur, mine, base, changed) {
        var same = base ? hash(cur) === base : cur === '';
        if (!changed) return { value: cur, write: false, sendBack: !same };
        if (!same) return { value: cur, write: false, conflict: true, sendBack: true };
        return { value: mine, write: true };
      }
      function saveDay(d, tasks, journal, opts) {
        if (!opts) return Promise.resolve(null);
        return current(d, ['tasks', 'journal']).then(function (cur) {
          if (!cur) return null;
          var t = col(cur.tasks, String(tasks || ''), opts.th, opts.ct), j = col(cur.journal, String(journal || ''), opts.jh, opts.cj), put = {};
          if (t.write && t.value !== cur.tasks) put.tasks = t.value;
          if (j.write && j.value !== cur.journal) put.journal = j.value;
          if (Object.keys(put).length) write(d, put, cur);
          var res = { th: hash(t.value), jh: hash(j.value), conflictT: !!t.conflict, conflictJ: !!j.conflict, build: '', release: '' };
          if (t.sendBack) res.tasks = t.value;
          if (j.sendBack) res.journal = j.value;
          return res;
        });
      }
      function savePad(f, d, html, base) {
        return current(d, [f]).then(function (cur) {
          if (!cur) return null;
          html = String(html || '');
          var c = col(cur[f], html, base, true), hk = HKEY[f], res = { build: '' };
          if (c.conflict) { res[hk] = hash(cur[f]); res.conflict = true; res[f] = cur[f]; return res; }
          if (html !== cur[f]) { var p = {}; p[f] = html; write(d, p, cur); }
          res[hk] = hash(html);
          return res;
        });
      }
      function scratch(d, html, base) {
        return current(d, ['tho']).then(function (cur) {
          if (!cur) return null;
          html = String(html || '');
          if (base && hash(cur.tho) !== base) return { conflict: true, html: cur.tho, h: hash(cur.tho) };
          if (html !== cur.tho) write(d, { tho: html }, cur);
          return { h: hash(html) };
        });
      }
      // Lay unconfirmed edits over a server answer, so a save still on its way isn't undone by an older copy.
      function overlay(name, r) {
        var m = un(), keys = Object.keys(m);
        if (!r || !keys.length) return r;
        if (name === 'getWorkspaceData' && Array.isArray(r.days)) {
          r.days.forEach(function (day) {
            FIELDS.forEach(function (f) { var u = m[day.date + '|' + f]; if (u) { day[f] = u.v; day[HKEY[f]] = hash(u.v); } });
          });
        } else if ((name === 'getSyncState' || name === 'runImports') && r.days && typeof r.days === 'object') {
          Object.keys(r.days).forEach(function (d) {
            var e = r.days[d];
            FIELDS.forEach(function (f) { if (m[d + '|' + f]) { delete e[f]; delete e[HKEY[f]]; } });
            if (!FIELDS.some(function (f) { return e[f] !== undefined; })) delete r.days[d];
          });
        }
        return r;
      }
      // Live: another device's save within today ± 7 → the normal sync merges it in. Also settles edits left from an
      // earlier session: in the database → forget them; the database kept something else → the leftover goes to Backup.
      function watchDays() {
        if (watching) return; watching = true;
        var w = win(), first = true;
        db.collection('ganttDays').where('date', '>=', w[0]).where('date', '<=', w[1]).onSnapshot({ includeMetadataChanges: true }, function (qs) {
          var changed = false;
          qs.docs.forEach(function (snap) {
            if (snap.metadata.fromCache || snap.metadata.hasPendingWrites) return;
            var x = snap.data(), d = x.date || snap.id, m = un();
            FIELDS.forEach(function (f) {
              var u = m[d + '|' + f], at = Number(x[f + 'At']) || 0;
              if (u && at !== u.at) { backup(d, f, u.v, 'older edit lost (newest wins)'); unDrop(d, f, u.at); }
              else if (u) unDrop(d, f, u.at);
              var k = d + '|' + f;
              if (!first && seen[k] !== at && x.dev !== dev) changed = true;
              seen[k] = at;
            });
          });
          first = false;
          if (changed && typeof window.fwSync === 'function') try { window.fwSync(false); } catch (e) {}
        }, function (e) { watching = false; console.warn('Days live sync stopped: ' + (e.code || e)); });
      }
      function editable(el) {
        if (el.closest('#scratchPad')) return true;
        var ed = el.closest('.day-editor[data-date],.journal-editor[data-date],.plan-pad[data-date],.thought-pad[data-date]');
        return !!ed && inWin(ed.getAttribute('data-date'));
      }
      return {
        watch: watchDays, overlay: overlay, editable: editable,
        run: function (name, a) {
          if (name === 'saveDay') return saveDay(a[0], a[1], a[2], a[3]);
          if (name === 'savePlanPad') return savePad('plan', a[0], a[1], a[2]);
          if (name === 'saveThoughtPad') return savePad('tho', a[0], a[1], a[2]);
          if (name === 'saveScratchPad') return scratch(a[0], a[1], a[2]);
          return current(a[0], ['tho']).then(function (cur) { return cur ? { html: cur.tho, h: hash(cur.tho) } : null; });
        }
      };
    }
    var days = DAYS ? daysArea() : null;
    /* ---------- Reference + Archive (H5 C, 2026-11-23): doc per card ganttRef/{id} ----------
       Same answers as the server's saveReferenceItem / addReferenceItem / saveReferenceOrder / saveReferenceExpanded /
       deleteReferenceItem / getArchive / restoreArchivedItem. A deleted card keeps its doc with del = deleted-at (that's
       the Archive). Online only (offline, Reference stays read-only); not seeded yet / card not in the database → the
       server. Newest wins: every write carries a newer `at`; a refused write → the losing copy to ganttBackup + a sync. */
    function refArea() {
      var seeded = false, watching = false, syncT = 0;
      function coll() { return db.collection('ganttRef'); }
      function rdoc(id) { return coll().doc(String(id)); }
      function ready() {
        if (!navigator.onLine) return Promise.resolve(false);
        if (seeded) return Promise.resolve(true);
        // Seeded by the server, and the rules let this account read cards (rules not published yet → refused → server).
        return db.collection('gantt').doc('refMeta').get({ source: 'server' }).then(function (s) {
          if (!(s.exists && s.data().seeded)) return false;
          return rdoc('__probe').get({ source: 'server' }).then(function () { return (seeded = true); });
        });
      }
      function one(id) { return rdoc(id).get({ source: 'server' }).then(function (s) { return s.exists ? s.data() : null; }); }
      function all() { return coll().get({ source: 'server' }).then(function (qs) { return qs.docs.map(function (s) { var d = s.data(); d.id = String(d.id || s.id); return d; }); }); }
      function stamp(rec, prevAt) { rec.at = Math.max(Date.now(), (Number(prevAt) || 0) + 1); rec.up = firebase.firestore.FieldValue.serverTimestamp(); rec.dev = dev; return rec; }
      function backup(id, html, reason) {
        db.collection('ganttBackup').add({ area: 'ref', card: String(id), html: String(html || ''), at: Date.now(), reason: reason, dev: dev })
          .catch(function (e) { console.warn('Backup not saved: ' + (e.code || e)); });
      }
      function lost(e) {
        if (e && e.code === 'permission-denied') { if (typeof window.fwSync === 'function') try { window.fwSync(false); } catch (x) {} }
        else console.warn('Reference not saved to the database: ' + (e && (e.code || e.message) || e));
      }
      function openTasks(h) { return (String(h || '').match(/class="check-line(?![^"]*\bdone\b)[^"]*"/g) || []).length; }
      function card(d) {
        var c = String(d.content || '');
        return { id: String(d.id), parentId: String(d.parentId || ''), type: String(d.type || 'item'), title: String(d.title || ''),
          content: c, expanded: Number(d.expanded) !== 0, sort: Number(d.sort) || 0, ref_: null, row: null, ch: hash(c) };
      }
      // Several docs, one batch per 400; a refusal (another device's newer copy) → write them one by one, re-read.
      function many(list) {
        for (var i = 0; i < list.length; i += 400) (function (part) {
          var b = db.batch();
          part.forEach(function (x) { b.set(rdoc(x.id), stamp(Object.assign({}, x.put), x.at), { merge: true }); });
          b.commit().catch(function (e) {
            if (!e || e.code !== 'permission-denied') return lost(e);
            part.forEach(function (x) { one(x.id).then(function (d) { if (d) return rdoc(x.id).set(stamp(Object.assign({}, x.put), d.at), { merge: true }); }).catch(lost); });
          });
        })(list.slice(i, i + 400));
      }
      function tree(docs, id) {
        var ids = {}, more = true; ids[id] = 1;
        while (more) { more = false; docs.forEach(function (d) { if (ids[String(d.parentId || '')] && !ids[d.id]) { ids[d.id] = 1; more = true; } }); }
        return docs.filter(function (d) { return ids[d.id]; });
      }
      function save(item) {
        if (!item || !item.fwSync || !item.id) return Promise.resolve(null);
        return one(item.id).then(function (d) {
          if (!d) return null; // not in the database: the server decides (new card, deleted elsewhere…)
          var res = { id: String(item.id), build: '', release: '' };
          if (d.del) { res.deleted = true; return res; }
          var cur = String(d.content || ''), mine = String(item.content || ''), content = cur, put = {};
          var same = item.baseHash ? hash(cur) === item.baseHash : item.baseText != null ? cur === String(item.baseText) : cur === '';
          if (!item.contentChanged) { if (!same) res.content = cur; }
          else if (!same) { res.conflict = true; res.content = cur; }
          else {
            content = mine;
            if (mine !== cur) {
              if (cur.length > 400 && mine.length < cur.length * 0.6) backup(item.id, cur, 'big shrink');
              else if (openTasks(mine) < openTasks(cur)) backup(item.id, cur, 'open task removed');
              put.content = mine; put.ch = hash(mine); put.base = hash(cur);
            }
          }
          res.meta = {}; res.metaConflicts = [];
          ['parentId', 'expanded'].forEach(function (key) {
            var norm = function (v) { return key === 'expanded' ? v !== false && v !== 0 : String(v || ''); };
            var now = key === 'expanded' ? Number(d.expanded) !== 0 : String(d.parentId || '');
            var ch = item.metaChanges && item.metaChanges[key];
            if (ch && 'before' in ch && 'after' in ch) {
              var b = norm(ch.before), a = norm(ch.after);
              if (now === b || now === a) { if (now !== a) put[key] = key === 'expanded' ? (a ? 1 : 0) : a; now = a; }
              else res.metaConflicts.push(key);
            }
            res.meta[key] = now;
          });
          if (Object.keys(put).length) rdoc(item.id).set(stamp(put, d.at), { merge: true }).catch(function (e) {
            if (e && e.code === 'permission-denied' && put.content !== undefined) backup(item.id, put.content, 'older edit lost (newest wins)');
            lost(e);
          });
          res.ch = hash(content);
          return res;
        });
      }
      function add(parentId, title, content) {
        var id = window.crypto && crypto.randomUUID ? crypto.randomUUID() : 'r' + Date.now().toString(36) + Math.random().toString(36).slice(2, 10);
        content = String(content || '');
        rdoc(id).set(stamp({ id: id, parentId: String(parentId || ''), type: 'item', title: String(title || ''), content: content, ch: hash(content),
          base: '', expanded: 1, sort: Date.now(), del: '' }, 0)).catch(lost);
        return Promise.resolve(id);
      }
      function archive(docs) {
        var rows = docs.filter(function (d) { return d.del; }), groups = {};
        rows.forEach(function (d) { (groups[d.del] = groups[d.del] || {})[d.id] = 1; });
        return rows.filter(function (d) { return !groups[d.del][String(d.parentId || '')]; })
          .sort(function (a, b) { return String(b.del).localeCompare(String(a.del)); })
          .map(function (d) {
            return { id: d.id, deletedAt: String(d.del), title: String(d.title || ''), content: String(d.content || ''),
              children: rows.filter(function (y) { return y.del === d.del && y.id !== d.id; }).length };
          });
      }
      function restore(id, deletedAt) {
        return all().then(function (docs) {
          var group = docs.filter(function (d) { return d.del === String(deletedAt); });
          var root = group.filter(function (d) { return d.id === String(id); })[0];
          if (!root) throw new Error('That cell is no longer in the archive.');
          var live = {}; docs.forEach(function (d) { if (!d.del) live[d.id] = 1; });
          var back = tree(group, root.id).filter(function (d) { return !live[d.id]; });
          many(back.map(function (d) {
            var put = { del: '' };
            if (d.id === root.id && d.parentId && !live[d.parentId]) put.parentId = '';
            Object.assign(d, put);
            return { id: d.id, put: put, at: d.at };
          }));
          return docs.filter(function (d) { return !d.del; }).map(card).sort(function (a, b) { return a.sort - b.sort; });
        });
      }
      function watchRef() {
        if (watching) return; watching = true;
        var first = true;
        coll().where('up', '>', new Date(Date.now() - 60000)).onSnapshot({ includeMetadataChanges: true }, function (qs) {
          if (first) { first = false; return; }
          var theirs = qs.docChanges().some(function (c) { return !c.doc.metadata.hasPendingWrites && c.doc.data().dev !== dev; });
          if (!theirs) return;
          clearTimeout(syncT);
          syncT = setTimeout(function () { if (typeof window.fwSync === 'function') try { window.fwSync(false); } catch (e) {} }, 400);
        }, function (e) { watching = false; console.warn('Reference live sync stopped: ' + (e.code || e)); });
      }
      return {
        watch: watchRef,
        run: function (name, a) {
          return ready().then(function (ok) {
            if (!ok) return null;
            if (name === 'saveReferenceItem') return save(a[0]);
            if (name === 'addReferenceItem') return add(a[0], a[1], a[2]);
            if (name === 'getArchive') return all().then(archive);
            if (name === 'restoreArchivedItem') return restore(a[0], a[1]);
            if (name === 'saveReferenceOrder') {
              if (!Array.isArray(a[0])) return true;
              return all().then(function (docs) {
                var at = {}; docs.forEach(function (d) { at[d.id] = d.at; });
                many(a[0].filter(function (e) { return at[String(e.id)] !== undefined; }).map(function (e) { return { id: String(e.id), put: { sort: Math.round(Number(e.sort) || 0) }, at: at[String(e.id)] }; }));
                return true;
              });
            }
            if (name === 'saveReferenceExpanded') return all().then(function (docs) {
              many(docs.filter(function (d) { return !d.del; }).map(function (d) { return { id: d.id, put: { expanded: a[0] !== false ? 1 : 0 }, at: d.at }; }));
              return true;
            });
            if (name === 'deleteReferenceItem') return all().then(function (docs) {
              var when = new Date().toISOString();
              many(tree(docs.filter(function (d) { return !d.del; }), String(a[0])).map(function (d) { return { id: d.id, put: { del: when }, at: d.at }; }));
              return true;
            });
            return null;
          });
        }
      };
    }
    var refs = REF ? refArea() : null;
    return {
      handles: function (name) { return !!NAMES[name]; },
      // Offline: typing allowed where the edit can be queued (Keep In Mind; days within today ± 7).
      editable: function (el) {
        if (!user || !el || !el.closest) return false;
        if (KIM && el.closest('#kimScroll')) return true;
        return !!(days && days.editable(el));
      },
      offlineText: function () {
        if (!user) return 'Offline · read-only';
        return 'Offline · only ' + (KIM && DAYS ? 'Keep In Mind + today ± 7 days save' : KIM ? 'Keep In Mind saves' : 'today ± 7 days save');
      },
      // Server answers that carry days: edits not yet in the database stay on top.
      overlay: function (name, r) {
        if (!days) return r;
        // batchCalls (speed plan E): each answer gets the same overlay as if it came alone.
        if (name === 'batchCalls' && Array.isArray(r) && Array.isArray(arguments[2])) {
          var list = arguments[2];
          r.forEach(function (o, i) { if (o && o.ok && list[i]) o.r = days.overlay(list[i][0], o.r); });
          return r;
        }
        return days.overlay(name, r);
      },
      run: function (name, args) {
        return withTimeout(boot, 4000).then(function () {
          if (!user) return null;
          if (DAY_NAMES[name]) return days.run(name, args);
          if (REF_NAMES[name]) return refs.run(name, args);
          return name === 'getKim' ? read() : save(args[0], args[1]);
        }).catch(function (e) { console.warn('Firestore → server: ' + (e && e.message || e)); return null; })
          .then(function (r) { return r || call(name, args); });
      }
    };
  })();

  /* ---------- google.script.run → fetch ---------- */
  function call(fn, args) {
    return fetch(API, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ k: key(), fn: fn, a: args })
    }).then(function (res) {
      if (!res.ok) throw new Error('Server error ' + res.status);
      return res.json();
    }, function () {
      setOffline(true);
      throw new Error('Offline (can’t reach the server)');
    }).then(function (j) {
      if (offline) setOffline(false);
      if (!j || !j.ok) {
        if (j && j.auth) { ls.del(KEY_LS); needKey('That key didn’t work.'); }
        throw new Error((j && j.e) || 'Server error');
      }
      return j.r;
    });
  }
  function runner(ok, bad, user) {
    return new Proxy({}, {
      get: function (_, name) {
        if (name === 'withSuccessHandler') return function (f) { return runner(f, bad, user); };
        if (name === 'withFailureHandler') return function (f) { return runner(ok, f, user); };
        if (name === 'withUserObject') return function (u) { return runner(ok, bad, u); };
        if (typeof name !== 'string') return undefined;
        return function () {
          var args = Array.prototype.slice.call(arguments);
          (fb.handles(name) ? fb.run(name, args) : call(name, args)).then(function (r) {
            if (name === 'getWorkspaceData' && PAGE === 'index') saveData(r);
            r = fb.overlay(name, r, args[0]);
            if (ok) ok(r, user);
          }, function (err) {
            if (bad) bad(err, user); else console.error(name + ': ' + err.message);
          });
        };
      }
    });
  }

  /* ---------- main page: open from the last data on this device ---------- */
  function today() {
    var d = new Date();
    return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
  }
  function saveData(r) {
    if (!r || !Array.isArray(r.days) || !r.today) return;
    try {
      var copy = Object.assign({}, r); delete copy.timing;
      if (!ls.set(DATA_LS, JSON.stringify({ at: Date.now(), data: copy }))) ls.del(DATA_LS);
    } catch (e) {}
  }
  window.__PWA_INITIAL__ = null;
  if (PAGE === 'index') {
    var cached = null;
    try { cached = JSON.parse(ls.get(DATA_LS) || 'null'); } catch (e) {}
    if (cached && cached.data && (cached.data.today === today() || offline)) {
      window.__PWA_INITIAL__ = fb.overlay('getWorkspaceData', cached.data);
      window.__PWA__.fromCache = cached.at;
      window.addEventListener('load', function () {
        // Catch up right away (days, Reference, imports); the stored copy refreshes a bit later.
        setTimeout(function () { if (!offline && typeof window.fwSync === 'function') try { window.fwSync(true); } catch (e) {} }, 50);
        setTimeout(refreshCopy, 20000);
      });
    } else {
      window.addEventListener('load', function () { setTimeout(refreshCopy, 20000); });
    }
  }
  function refreshCopy() {
    if (offline || document.hidden) return;
    call('getWorkspaceData', [{ fast: true }]).then(saveData, function () {});
  }

  /* ---------- service worker + update button ---------- */
  if ('serviceWorker' in navigator) {
    // Reload only when "↻ Update ready" was tapped; the first install also changes the controller (clients.claim),
    // and reloading then made the page jump on open (2026-11-21).
    var wantReload = false, reloading = false;

      if (!wantReload || reloading) return; reloading = true; location.reload();
    });

            try { if (typeof window.fwFlushAll === 'function') window.fwFlushAll(); } catch (e) {}
            setTimeout(function () { wantReload = true; w.postMessage('skipWaiting'); }, 1200);
          };
          document.body.appendChild(b);
        }
        if (reg.waiting) offer(reg.waiting);
        reg.addEventListener('updatefound', function () {
          var w = reg.installing;
          if (w) w.addEventListener('statechange', function () { if (w.state === 'installed') offer(w); });
        });
        var check = function () { if (!document.hidden && navigator.onLine) reg.update().catch(function () {}); };
        document.addEventListener('visibilitychange', check);
        setInterval(check, 10 * 60 * 1000);
      }).catch(function (e) { console.warn('SW', e); });
    });
  }
