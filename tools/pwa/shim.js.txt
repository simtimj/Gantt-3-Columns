    // cellHash_ (Code.gs): first 8 bytes of the MD5 of the UTF-8 text, as hex. Same versions as the server's.
    function hash(text) {
      var b = new TextEncoder().encode(String(text || '')), n = b.length, words = new Array(((n + 8) >> 6) + 1 << 4).fill(0), i;
      for (i = 0; i < n; i++) words[i >> 2] |= b[i] << (i % 4 * 8);
      words[n >> 2] |= 0x80 << (n % 4 * 8);
      words[words.length - 2] = n * 8;
      var S = [7, 12, 17, 22, 5, 9, 14, 20, 4, 11, 16, 23, 6, 10, 15, 21], K = [];
      for (i = 0; i < 64; i++) K[i] = Math.floor(Math.abs(Math.sin(i + 1)) * 4294967296) | 0;
      var h = [0x67452301, 0xefcdab89 | 0, 0x98badcfe | 0, 0x10325476];
      for (var o = 0; o < words.length; o += 16) {
        var a = h[0], c = h[2], d = h[3], bb = h[1];
        for (i = 0; i < 64; i++) {
          var f, g, r = i >> 4;
          if (r === 0) { f = (bb & c) | (~bb & d); g = i; }
          else if (r === 1) { f = (d & bb) | (~d & c); g = (5 * i + 1) % 16; }
          else if (r === 2) { f = bb ^ c ^ d; g = (3 * i + 5) % 16; }
          else { f = c ^ (bb | ~d); g = (7 * i) % 16; }
          var t = d; d = c; c = bb;
          var x = (a + f + K[i] + words[o + g]) | 0, sh = S[r * 4 + i % 4];
          bb = (bb + ((x << sh) | (x >>> (32 - sh)))) | 0; a = t;
        }
        h[0] = (h[0] + a) | 0; h[1] = (h[1] + bb) | 0; h[2] = (h[2] + c) | 0; h[3] = (h[3] + d) | 0;
      }
      var out = '';
      for (i = 0; i < 8; i++) out += ((h[i >> 2] >>> (i % 4 * 8)) & 255).toString(16).padStart(2, '0');
      return out;
    }
    /* ---------- Days (H4): doc per day ganttDays/{date}, newest wins per section ----------
       Same answers as the server's saveDay / savePlanPad / saveThoughtPad / saveScratchPad / getScratchPad, with
       the same hashes (MD5 below = cellHash_). A section not in the database yet: online → the server (it writes the
       doc); offline → checked against this device's last copy. Unconfirmed edits stay in localStorage and are laid
       over server answers and the opening data until the database has them. */
    function daysArea() {
      var FIELDS = ['tasks', 'journal', 'plan', 'tho'], HKEY = { tasks: 'th', journal: 'jh', plan: 'ph', tho: 'hh' };
      var DU = 'pwa.days.unconfirmed', WIN = 7, seen = {}, watching = false;
      function dayDoc(d) { return db.collection('ganttDays').doc(d); }

      function win() { var n = Date.now(); return [ymd(n - WIN * 864e5), ymd(n + WIN * 864e5)]; }
      function inWin(d) { var w = win(); return !!d && d >= w[0] && d <= w[1]; }
      function un() { try { return JSON.parse(ls.get(DU) || '{}') || {}; } catch (e) { return {}; } }
      function unSet(m) { if (Object.keys(m).length) ls.set(DU, JSON.stringify(m)); else ls.del(DU); }
      function unPut(d, f, v, at) { var m = un(); m[d + '|' + f] = { v: v, at: at }; unSet(m); }
      function unDrop(d, f, at) { var m = un(), k = d + '|' + f; if (m[k] && (at == null || m[k].at === at)) { delete m[k]; unSet(m); } }
      // This device's last copy of a section (offline check): an unconfirmed edit, else the stored opening data.
      function localCopy(d, f) {
        var u = un()[d + '|' + f]; if (u) return u.v;
        try {
          var c = JSON.parse(ls.get(DATA_LS) || 'null'), day = c && c.data && (c.data.days || []).filter(function (x) { return x.date === d; })[0];
          return day && day[f] != null ? String(day[f]) : null;
        } catch (e) { return null; }
      }
      // A day's doc (null = none). known = the answer is certain (from the database, or a cached "no such doc").
      function readDoc(d) {
        var cache = function () { return dayDoc(d).get({ source: 'cache' }).catch(function () { return null; }); };
        return (navigator.onLine ? dayDoc(d).get({ source: 'server' }).catch(function (e) { if (refused(e)) throw e; return cache(); }) : cache()).then(function (snap) {
          return { doc: snap && snap.exists ? snap.data() : null, known: !!snap };
        });
      }
      // Current copy of each wanted section, or null when one isn't known here (→ server path). After the full
      // history seed (H5 D) a day with no doc is an empty day, and a doc's missing section is empty too.
      function current(d, fields) {
        return Promise.all([readDoc(d), seededMeta()]).then(function (x) {
          var doc = x[0].doc, full = !!x[1] && x[0].known;
          var cur = { _doc: doc };
          for (var i = 0; i < fields.length; i++) {
            var f = fields[i];
            if (doc && doc[f + 'At']) cur[f] = String(doc[f] == null ? '' : doc[f]);
            else if (full) cur[f] = '';
            else if (navigator.onLine) return null;
            else { var lc = localCopy(d, f); if (lc == null) return null; cur[f] = lc; }
          }
          return cur;
        });
      }
      function backup(d, f, html, reason) {
        db.collection('ganttBackup').add({ area: 'days', date: d, field: f, html: String(html || ''), at: Date.now(), reason: reason, dev: dev })
          .catch(function (e) { console.warn('Backup not saved: ' + (e.code || e)); });
      }
      function openTasks(h) { return (String(h || '').match(/class="check-line(?![^"]*\bdone\b)[^"]*"/g) || []).length; }
      // Write the given sections (not awaited: offline it resolves only once the queued edit reaches the database).
      function write(d, put, cur) {
        var doc = cur._doc || {}, at = Date.now(), rec = { date: d, dev: dev, up: firebase.firestore.FieldValue.serverTimestamp() };
        Object.keys(put).forEach(function (f) { at = Math.max(at, (Number(doc[f + 'At']) || 0) + 1); });
        rec.at = at;
        Object.keys(put).forEach(function (f) {
          var old = cur[f];
          if (old.length > 400 && put[f].length < old.length * 0.6) backup(d, f, old, 'big shrink');
          else if (f === 'tasks' || f === 'journal') { if (openTasks(put[f]) < openTasks(old)) backup(d, f, old, 'open task removed'); }
          rec[f] = put[f]; rec[f + 'At'] = at; rec[f + 'Base'] = hash(old);
          unPut(d, f, put[f], at);
        });
        dayDoc(d).set(rec, { merge: true }).then(function () {
          Object.keys(put).forEach(function (f) { unDrop(d, f, at); });
        }, function (e) {
          if (e && e.code === 'permission-denied') {
            Object.keys(put).forEach(function (f) { backup(d, f, put[f], 'older edit lost (newest wins)'); unDrop(d, f, at); });
            if (typeof window.fwSync === 'function') try { window.fwSync(false); } catch (x) {}
          } else console.warn('Day not saved to the database: ' + (e.code || e));
        });
      }
      // The server's rule for one section (syncCol_ / the pads): same as the copy the app started from → write.
      function col(cur, mine, base, changed) {
        var same = base ? hash(cur) === base : cur === '';
        if (!changed) return { value: cur, write: false, sendBack: !same };
        if (!same) return { value: cur, write: false, conflict: true, sendBack: true };
        return { value: mine, write: true };
      }
      function saveDay(d, tasks, journal, opts) {
        if (!opts) return Promise.resolve(null);
        return current(d, ['tasks', 'journal']).then(function (cur) {
          if (!cur) return null;
          var t = col(cur.tasks, String(tasks || ''), opts.th, opts.ct), j = col(cur.journal, String(journal || ''), opts.jh, opts.cj), put = {};
          if (t.write && t.value !== cur.tasks) put.tasks = t.value;
          if (j.write && j.value !== cur.journal) put.journal = j.value;
          if (Object.keys(put).length) write(d, put, cur);
          var res = { th: hash(t.value), jh: hash(j.value), conflictT: !!t.conflict, conflictJ: !!j.conflict, build: '', release: '' };
          if (t.sendBack) res.tasks = t.value;
          if (j.sendBack) res.journal = j.value;
          return res;
        });
      }
      function savePad(f, d, html, base) {
        return current(d, [f]).then(function (cur) {
          if (!cur) return null;
          html = String(html || '');
          var c = col(cur[f], html, base, true), hk = HKEY[f], res = { build: '' };
          if (c.conflict) { res[hk] = hash(cur[f]); res.conflict = true; res[f] = cur[f]; return res; }
          if (html !== cur[f]) { var p = {}; p[f] = html; write(d, p, cur); }
          res[hk] = hash(html);
          return res;
        });
      }
      function scratch(d, html, base) {
        return current(d, ['tho']).then(function (cur) {
          if (!cur) return null;
          html = String(html || '');
          if (base && hash(cur.tho) !== base) return { conflict: true, html: cur.tho, h: hash(cur.tho) };
          if (html !== cur.tho) write(d, { tho: html }, cur);
          return { h: hash(html) };
        });
      }
      /* ---------- Full history (H5 D, 2026-11-23) ----------
         The server seeds every Sheet day into ganttDays once (gantt/daysMeta = { seeded, first, last, days }). This
         device keeps its own copy (Firestore's offline cache) up to date with one small query: days whose `up` is past

    /* ---------- Calendar copy (H5 B2, 2026-11-23) ----------

    function calArea() {
      var meta = null, months = {}, off = false, dirtyAt = 0, watching = false, gotFirst, first = new Promise(function (ok) { gotFirst = ok; });
      var DAYMS = 864e5;
      function list(x) { try { var v = JSON.parse(x || '[]'); return Array.isArray(v) ? v : []; } catch (e) { return []; } }
      function watch() {
        if (watching) return; watching = true;
        try {
          db.collection('ganttCal').onSnapshot(function (qs) {
            qs.docChanges().forEach(function (c) {
              var id = c.doc.id, d = c.doc.data() || {};
              if (c.type === 'removed') { if (id === 'meta') meta = null; else delete months[id]; return; }
              if (id === 'meta') meta = d;
              else if (/^\d{4}-\d{2}$/.test(id)) months[id] = list(d.ev);
            });

            gotFirst();

        } catch (e) { off = true; gotFirst(); }
      }
      function addDay(d, n) { return new Date(Date.parse(d + 'T00:00:00Z') + n * DAYMS).toISOString().slice(0, 10); }
      // Events overlapping [s, e) (yyyy-MM-dd, e exclusive) from the copy, or null → server.
      function pick(s, e) {
        if (off || !meta || Number(meta.v) !== 1) return null;
        if (dirtyAt && Date.now() - dirtyAt < 90000) return null;
        if (navigator.onLine && Date.now() - Number(meta.at) > 20 * 60000) return null;
        if (!/^\d{4}-\d{2}-\d{2}$/.test(s) || !/^\d{4}-\d{2}-\d{2}$/.test(e) || e <= s || s < meta.from || e > meta.to) return null;
        var have = list(meta.months), seen = {}, out = [];
        for (var m = s.slice(0, 7), last = addDay(e, -1).slice(0, 7); m <= last; m = addDay(m + '-28', 7).slice(0, 7)) {
          if (have.indexOf(m) < 0 || !months[m]) return null;
          months[m].forEach(function (ev) {
            if (seen[ev.id]) return;
            var hit = ev.allDay ? ev.start < e && ev.end > s : String(ev.start).slice(0, 16) < e + 'T00:00' && String(ev.end).slice(0, 16) > s + 'T00:00';
            if (hit) { seen[ev.id] = 1; out.push(ev); }
          });
        }
        return out;
      }
      function clean(ev) { var o = Object.assign({}, ev); delete o.uc; return o; }
      // Local midnight (America/Los_Angeles) of a date, in ms, like the server's parseDate_.
      function midnight(d) {

        try {
          var t = new Intl.DateTimeFormat('en-US', { timeZone: meta.tz || 'America/Los_Angeles', timeZoneName: 'shortOffset' }).formatToParts(probe)
            .filter(function (p) { return p.type === 'timeZoneName'; })[0].value, mm = t.match(/GMT([+-])(\d+)(?::(\d+))?/);
          if (mm) mins = (mm[1] === '-' ? -1 : 1) * (Number(mm[2]) * 60 + Number(mm[3] || 0));
        } catch (e) {}
        return Date.parse(d + 'T00:00:00Z') - mins * 60000;
      }
      function hm(iso) { var h = Number(String(iso).slice(11, 13)), m = String(iso).slice(14, 16); return (h % 12 || 12) + ':' + m + ' ' + (h < 12 ? 'AM' : 'PM'); }

              time: ev.allDay ? 'All day' : hm(ev.start) + '–' + hm(ev.end) };

          });

        }
      };
      return {
        watch: watch,
        run: function (name, a) { return withTimeout(first, 3000).catch(function () {}).then(function () { return RUN[name].apply(null, a || []); }); },

      };

          if (!item.contentChanged) { if (!same) res.content = cur; }

            if (ch && 'before' in ch && 'after' in ch) {

              else res.metaConflicts.push(key);

      function add(parentId, title, content) {

      function archive(docs) {
        var rows = docs.filter(function (d) { return d.del; }), groups = {};
        rows.forEach(function (d) { (groups[d.del] = groups[d.del] || {})[d.id] = 1; });
        return rows.filter(function (d) { return !groups[d.del][String(d.parentId || '')]; })
          .sort(function (a, b) { return String(b.del).localeCompare(String(a.del)); })
          .map(function (d) {
            return { id: d.id, deletedAt: String(d.del), title: String(d.title || ''), content: String(d.content || ''),
              children: rows.filter(function (y) { return y.del === d.del && y.id !== d.id; }).length };
          });
      }
      function restore(id, deletedAt) {
        return all().then(function (docs) {
          var group = docs.filter(function (d) { return d.del === String(deletedAt); });
          var root = group.filter(function (d) { return d.id === String(id); })[0];
          if (!root) throw new Error('That cell is no longer in the archive.');
          var live = {}; docs.forEach(function (d) { if (!d.del) live[d.id] = 1; });
          var back = tree(group, root.id).filter(function (d) { return !live[d.id]; });
          many(back.map(function (d) {
            var put = { del: '' };
            if (d.id === root.id && d.parentId && !live[d.parentId]) put.parentId = '';
            Object.assign(d, put);
            return { id: d.id, put: put, at: d.at };
          }));
          return docs.filter(function (d) { return !d.del; }).map(card).sort(function (a, b) { return a.sort - b.sort; });
        });
      }
      function watchRef() {

      }

                many(a[0].filter(function (e) { return at[String(e.id)] !== undefined; }).map(function (e) { return { id: String(e.id), put: { sort: Math.round(Number(e.sort) || 0) }, at: at[String(e.id)] }; }));
                return true;
              });
            }
            if (name === 'saveReferenceExpanded') return all().then(function (docs) {

              return true;
            });
            if (name === 'deleteReferenceItem') return all().then(function (docs) {
              var when = new Date().toISOString();
              many(tree(docs.filter(function (d) { return !d.del; }), String(a[0])).map(function (d) { return { id: d.id, put: { del: when }, at: d.at }; }));

    var refs = REF ? refArea() : null;
    /* ---------- Opening from the database (H5 D) ----------
       Same answer as the server's getWorkspaceData({ fast }): days (today −7 … +3, with hashes), Reference (live
       cards), Urgent (every day with an urgency or due tag), Notes seeds (newest must / focus section), plus the
       card categories and pins from the database (H5 F; before its seed: this device's last opening; the sync right after the

    var PAST = 7, FUTURE = 3;
    function addDays(d, n) { var t = new Date(Date.UTC(+d.slice(0, 4), +d.slice(5, 7) - 1, +d.slice(8, 10)) + n * 864e5); return t.toISOString().slice(0, 10); }
    function opening(opts) {
      if (!days || !refs || !(opts && opts.fast) || !navigator.onLine) return Promise.resolve(null);
      var last = null;
      try { last = JSON.parse(ls.get(DATA_LS) || 'null'); } catch (e) {}
      last = last && last.data;

        var by = x[0], cards = x[1], set = x[2] && x[2].has ? x[2] : null;
        // Pins + card categories: from the database (H5 F), else this device's last opening, else the server.
        if (!set) { if (!last || !last.cardCats) return null; set = { cardCats: last.cardCats, refPins: last.refPins || [] }; }
        if (!by || !cards) return null;
        var today = days.ymd(Date.now()), out = days.build(by, addDays(today, -PAST), addDays(today, FUTURE));
        if (!out) return null;
        var t1 = Date.now();

          return { ref_: null, id: String(d.id), parentId: String(d.parentId || ''), type: String(d.type || 'item'), title: String(d.title || ''),
            content: String(d.content || ''), expanded: Number(d.expanded) !== 0, sort: Number(d.sort) || 0, row: null, ch: hash(String(d.content || '')) };
        }).sort(function (a, b) { return a.sort - b.sort; });
        var manifest = reference.map(function (r) { return { id: r.id, parentId: r.parentId, type: r.type, title: r.title, expanded: r.expanded, sort: r.sort }; });
        var urgent = [], seeds = {}, has = function (h) { return h.indexOf('data-urg=') >= 0 || h.indexOf('data-due=') >= 0; };
        Object.keys(by).sort().forEach(function (date) {
          var d = by[date], tasks = String(d.tasks || ''), journal = String(d.journal || ''), plan = String(d.plan || '');
          if (has(tasks) || has(journal) || has(plan)) urgent.push({ date: date, tasks: has(tasks) ? tasks : '', journal: has(journal) ? journal : '', plan: has(plan) ? plan : '' });
          if (date <= today && journal.indexOf('data-nsec') >= 0) ['must', 'focus'].forEach(function (k) {
            if (new RegExp('data-nsec=["\']' + k + '["\']').test(journal)) seeds[k] = { date: date, journal: journal };
          });
        });
        return { days: out, reference: reference, refManifestHash: hash(JSON.stringify(manifest)), urgent: urgent, inbox: [],
          cardCats: set.cardCats, refPins: set.refPins, notesSeeds: seeds, today: today,
          timing: { total: Date.now() - t0, steps: { dbRead: t1 - t0, build: Date.now() - t1 }, release: '', fast: true, fromDb: true,
            dayRows: Object.keys(by).length, refRows: reference.length } };
      });
    }
    return {
      handles: function (name) { return !!NAMES[name]; },
      // After a successful call: calendar edits mark the copy out of date (B2).
      after: function (name) { if (cal && CAL_WRITES[name]) cal.touch(); },
      // Offline: typing allowed where the edit can be queued (Keep In Mind; days within today ± 7).
      editable: function (el) {

      },

        }

      },
      run: function (name, args) {

          if (CAL_NAMES[name]) return cal.run(name, args);

  /* ---------- google.script.run → fetch ---------- */
  function call(fn, args) {
    return fetch(API, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ k: key(), fn: fn, a: args })
    }).then(function (res) {
      if (!res.ok) throw new Error('Server error ' + res.status);
      return res.json();
    }, function () {
      setOffline(true);
      throw new Error('Offline (can’t reach the server)');
    }).then(function (j) {
      if (offline) setOffline(false);
      if (!j || !j.ok) {
        if (j && j.auth) { ls.del(KEY_LS); needKey('That key didn’t work.'); }
        throw new Error((j && j.e) || 'Server error');
      }
      return j.r;
    });
  }
  function runner(ok, bad, user) {
    return new Proxy({}, {
      get: function (_, name) {
        if (name === 'withSuccessHandler') return function (f) { return runner(f, bad, user); };
        if (name === 'withFailureHandler') return function (f) { return runner(ok, f, user); };
        if (name === 'withUserObject') return function (u) { return runner(ok, bad, u); };
        if (typeof name !== 'string') return undefined;
        return function () {
          var args = Array.prototype.slice.call(arguments);
          (fb.handles(name) ? fb.run(name, args) : call(name, args)).then(function (r) {
            if (name === 'getWorkspaceData' && PAGE === 'index') saveData(r);
            if (fb.after) fb.after(name);
            r = fb.overlay(name, r, args[0]);
            if (ok) ok(r, user);
          }, function (err) {
            if (bad) bad(err, user); else console.error(name + ': ' + err.message);
          });
        };
      }
    });
  }
