    // cellHash_ (Code.gs): first 8 bytes of the MD5 of the UTF-8 text, as hex. Same versions as the server's.
    function hash(text) {
      var b = new TextEncoder().encode(String(text || '')), n = b.length, words = new Array(((n + 8) >> 6) + 1 << 4).fill(0), i;
      for (i = 0; i < n; i++) words[i >> 2] |= b[i] << (i % 4 * 8);
      words[n >> 2] |= 0x80 << (n % 4 * 8);
      words[words.length - 2] = n * 8;
      var S = [7, 12, 17, 22, 5, 9, 14, 20, 4, 11, 16, 23, 6, 10, 15, 21], K = [];
      for (i = 0; i < 64; i++) K[i] = Math.floor(Math.abs(Math.sin(i + 1)) * 4294967296) | 0;
      var h = [0x67452301, 0xefcdab89 | 0, 0x98badcfe | 0, 0x10325476];
      for (var o = 0; o < words.length; o += 16) {
        var a = h[0], c = h[2], d = h[3], bb = h[1];
        for (i = 0; i < 64; i++) {
          var f, g, r = i >> 4;
          if (r === 0) { f = (bb & c) | (~bb & d); g = i; }
          else if (r === 1) { f = (d & bb) | (~d & c); g = (5 * i + 1) % 16; }
          else if (r === 2) { f = bb ^ c ^ d; g = (3 * i + 5) % 16; }
          else { f = c ^ (bb | ~d); g = (7 * i) % 16; }
          var t = d; d = c; c = bb;
          var x = (a + f + K[i] + words[o + g]) | 0, sh = S[r * 4 + i % 4];
          bb = (bb + ((x << sh) | (x >>> (32 - sh)))) | 0; a = t;
        }
        h[0] = (h[0] + a) | 0; h[1] = (h[1] + bb) | 0; h[2] = (h[2] + c) | 0; h[3] = (h[3] + d) | 0;
      }
      var out = '';
      for (i = 0; i < 8; i++) out += ((h[i >> 2] >>> (i % 4 * 8)) & 255).toString(16).padStart(2, '0');
      return out;
    }

    /* ---------- Calendar copy (H5 B2, 2026-11-23) ----------

    function calArea() {

      var DAYMS = 864e5;
      function list(x) { try { var v = JSON.parse(x || '[]'); return Array.isArray(v) ? v : []; } catch (e) { return []; } }
      function watch() {
        if (watching) return; watching = true;

          db.collection('ganttCal').onSnapshot(function (qs) {
            qs.docChanges().forEach(function (c) {
              var id = c.doc.id, d = c.doc.data() || {};

          }, function (e) { off = true; gotFirst(); console.warn('Calendar copy → server: ' + (e && e.code || e)); });
        } catch (e) { off = true; gotFirst(); }
      }
      /* After an edit the copy counts only once it holds the re-copy that edit asked for (calSyncNow answers with its
         meta `at`, server clock): a 5-min copy that read the calendar before the edit can finish after it and must
         not count (it would put a just-dragged event back). Re-copy failed → server path for 6 min (the trigger catches up). */
      function dirty() {
        if (!dirtyAt) return false;
        if (wantAt && meta && Number(meta.at) >= wantAt) { dirtyAt = 0; wantAt = 0; return false; }
        if (!wantAt && Date.now() - dirtyAt > 6 * 60000) { dirtyAt = 0; return false; }
        return true;
      }
      function caught() { if (dirtyAt) dirty(); }
      function addDay(d, n) { return new Date(Date.parse(d + 'T00:00:00Z') + n * DAYMS).toISOString().slice(0, 10); }
      // Events overlapping [s, e) (yyyy-MM-dd, e exclusive) from the copy, or null → server.
      function pick(s, e) {
        if (off || !meta || Number(meta.v) !== 1) return null;
        if (dirty()) return null;
        if (navigator.onLine && Date.now() - Number(meta.at) > 20 * 60000) return null;
        if (!/^\d{4}-\d{2}-\d{2}$/.test(s) || !/^\d{4}-\d{2}-\d{2}$/.test(e) || e <= s || s < meta.from || e > meta.to) return null;
        var have = list(meta.months), seen = {}, out = [];

          if (have.indexOf(m) < 0 || !months[m]) return null;
          months[m].forEach(function (ev) {
            if (seen[ev.id]) return;

        },
        run: function (name, a) { return withTimeout(first, 3000).catch(function () {}).then(function () { return RUN[name].apply(null, a || []); }); },

        settle: function (at) { ++touchN; dirtyAt = Date.now(); wantAt = Number(at) || 0; caught(); },

        if(same(t,b) || same(m,t)) return m;
        if(m === undefined && object(t)) return t;
        if(t === undefined && object(m)) return m;
        if(object(b) && object(m) && object(t)) {
          const newer = stamp(m) || stamp(t) ? stamp(m) >= stamp(t) : mineNewer;
          const out = {};
          [...new Set(Object.keys(b).concat(Object.keys(m),Object.keys(t)))].forEach(k => {
            const v = merge(b[k],m[k],t[k],path + '.' + k,newer);
            if(v !== undefined) out[k] = v;
          });
          return out;
        }
        const records = a => Array.isArray(a) && a.every(x => object(x) && typeof x.id === 'string');
        const uniq = a => { const seen = {}; return a.filter(x => seen[x.id] ? false : (seen[x.id] = true)); };
        if(Array.isArray(b) && records(m) && records(t) && (records(b) || !b.length)) {
          b = uniq(b); m = uniq(m); t = uniq(t);
          const maps = [b,m,t].map(a => new Map(a.map(x => [x.id,x]))), out = new Map();
          [...new Set(b.concat(m,t).map(x => x.id))].forEach(id => {
            const v = merge(maps[0].get(id),maps[1].get(id),maps[2].get(id),path + '[' + id + ']',mineNewer);
            if(v !== undefined) out.set(id,v);
          });
          const common = b.map(x => x.id).filter(id => maps[1].has(id) && maps[2].has(id) && out.has(id));
          const order = a => a.map(x => x.id).filter(id => common.indexOf(id) >= 0);
          const tc = !same(order(t),common) && same(order(m),common);
          const primary = tc ? t : m, secondary = tc ? m : t;
          const ids = primary.map(x => x.id).filter(id => out.has(id));
          secondary.forEach((x,i) => {
            if(!out.has(x.id) || ids.indexOf(x.id) >= 0) return;
            const next = secondary.slice(i+1).find(n => ids.indexOf(n.id) >= 0);
            ids.splice(next ? ids.indexOf(next.id) : ids.length,0,x.id);
          });
          out.forEach((v,id) => { if(ids.indexOf(id) < 0) ids.push(id); });
          return ids.map(id => out.get(id));
        }
        const win = mineNewer === false ? t : m, lose = win === m ? t : m;
        overwritten.push({path:path,lost:lose});
        return win;
      }
      return {data:merge(base,mine,theirs,'focus',true),conflicts:[],overwritten:overwritten};
    }

        var res = mergeFocus(base, data, cur), txt = JSON.stringify(res.data);

          });

      },

        if (KIM) parts.push('Keep In Mind'); if (focus) parts.push('Focus'); if (inbox) parts.push('Inbox'); if (DAYS) parts.push('today ± 7 days');
        return 'Offline · only ' + parts.join(' + ') + ' save' + (parts.length === 1 && !DAYS ? 's' : '');

      overlay: function (name, r) {

          r.forEach(function (o, i) { if (o && o.ok && list[i]) o.r = days.overlay(list[i][0], o.r); });

          if (r) { fromDb[name] = 1; return r; }
          return call(name, args);
        });
        return dbRun(name, args).then(function (r) {
          if (r) { fromDb[name] = 1; return r; }
          return call(name, args).then(function (x) { if (name === 'getSyncState') lastSrvSync = Date.now(); return x; });
        });
      }
    };
  })();

  /* ---------- google.script.run → fetch ---------- */
  function call(fn, args) {
    return fetch(API, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ k: key(), fn: fn, a: args })
    }).then(function (res) {
      if (!res.ok) throw new Error('Server error ' + res.status);
      return res.json();
    }, function () {
      setOffline(true);
      throw new Error('Offline (can’t reach the server)');
    }).then(function (j) {
      if (offline) setOffline(false);
      if (!j || !j.ok) {
        if (j && j.auth) { ls.del(KEY_LS); needKey('That key didn’t work.'); }
        throw new Error((j && j.e) || 'Server error');
      }
      return j.r;
    });
  }
  function runner(ok, bad, user) {
    return new Proxy({}, {
      get: function (_, name) {
        if (name === 'withSuccessHandler') return function (f) { return runner(f, bad, user); };
        if (name === 'withFailureHandler') return function (f) { return runner(ok, f, user); };
        if (name === 'withUserObject') return function (u) { return runner(ok, bad, u); };
        if (typeof name !== 'string') return undefined;
        return function () {
          var args = Array.prototype.slice.call(arguments);
          (fb.handles(name) ? fb.run(name, args) : call(name, args)).then(function (r) {
            if (name === 'getWorkspaceData' && PAGE === 'index') saveData(r);
            if (fb.after) fb.after(name);
            r = fb.overlay(name, r, args[0]);
            if (ok) ok(r, user);
          }, function (err) {
            if (bad) bad(err, user); else console.error(name + ': ' + err.message);
          });
        };
      }
    });
  }

  /* ---------- main page: open from the last data on this device ---------- */
  function today() {
    var d = new Date();
    return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
  }
  function saveData(r) {
    if (!r || !Array.isArray(r.days) || !r.today) return;
    try {
      var copy = Object.assign({}, r); delete copy.timing;
      if (!ls.set(DATA_LS, JSON.stringify({ at: Date.now(), data: copy }))) ls.del(DATA_LS);
    } catch (e) {}
  }
  window.__PWA_INITIAL__ = null;
  if (PAGE === 'index') {
    var cached = null;
    try { cached = JSON.parse(ls.get(DATA_LS) || 'null'); } catch (e) {}
    // Editable local-first (H5 G, 2026-11-23): a copy from an earlier day (≤ 7 days) is re-dated to today and opened
    // editable at once, like a same-day copy; the sync right after brings in every day that changed elsewhere.
    var fresh = cached && cached.data && Array.isArray(cached.data.days) ? redate(cached.data, today()) : null;
    if (fresh) window.__PWA__.redated = cached.data.today;
    if (fresh || (cached && cached.data && offline)) {
      window.__PWA_INITIAL__ = fb.overlay('getWorkspaceData', fresh || cached.data);
      window.__PWA__.fromCache = cached.at;
      window.addEventListener('load', function () {
        // Catch up right away (days, Reference, imports); the stored copy refreshes a bit later.
        setTimeout(function () { if (!offline && typeof window.fwSync === 'function') try { window.fwSync(true); } catch (e) {} }, 50);
        setTimeout(refreshCopy, 20000);
      });
    } else {
      window.addEventListener('load', function () { setTimeout(refreshCopy, 20000); });
    }
  }
  // The device's last opening, moved to `day`: days day − 7 … day + 3 (dates the copy lacks start empty, with the
  // empty-text hash, so the sync fetches them if they have text). Same day → as is; older than 7 days / future → null.
  function redate(data, day) {
    var EMPTY_H = 'd41d8cd98f00b204'; // cellHash_('') (MD5, first 16 hex)
    if (!data.today || data.today === day) return data;
    var ms = function (d) { return Date.parse(d + 'T00:00:00Z'); }, gap = (ms(day) - ms(data.today)) / 864e5;
    if (!(gap > 0 && gap <= 7)) return null;
    var by = {}, out = Object.assign({}, data), days = [];
    data.days.forEach(function (d) { by[d.date] = d; });
    for (var i = -7; i <= 3; i++) {
      var d = new Date(ms(day) + i * 864e5).toISOString().slice(0, 10);
      days.push(by[d] || { date: d, tasks: '', journal: '', plan: '', tho: '', th: EMPTY_H, jh: EMPTY_H, ph: EMPTY_H, hh: EMPTY_H });
    }
    out.days = days; out.today = day;
    return out;
  }

  if ('serviceWorker' in navigator) {
    // Reload only when "↻ Update ready" was tapped; the first install also changes the controller (clients.claim),

    });

             (a 2 s day save was still in flight, so the new page started from an old copy and its first save came
             back `conflict`); (2) reload once the new worker is fully active, not at controllerchange (the reload's
             page fetch sat ~1,000 ms before the worker answered: workerStart 1000). The old page stays on screen
             during the wait instead of a blank reload. */
          b.onclick = function () {

            try { if (typeof window.fwFlushAll === 'function') window.fwFlushAll(); } catch (e) {}
            var t0 = Date.now();
            var go = function () {
              try { sessionStorage.setItem('fw.updTap', String(t0)); } catch (e) {}
              b.textContent = 'Updating…';
              var done = function () { if (reloading) return; reloading = true; location.reload(); };
              w.addEventListener('statechange', function () { if (w.state === 'activated') done(); });
              wantReload = true; w.postMessage('skipWaiting');
              setTimeout(function () { wantReload = true; if (navigator.serviceWorker.controller === w || w.state === 'activated') done(); }, 4000);
            };

              if (busy && Date.now() - t0 < 10000) { setTimeout(wait, 150); return; }

            };
            setTimeout(wait, 50);
          };
          document.body.appendChild(b);
        }
        if (reg.waiting) offer(reg.waiting);

        });

        document.addEventListener('visibilitychange', check);
        setInterval(check, 10 * 60 * 1000);
      }).catch(function (e) { console.warn('SW', e); });
    });
  }
