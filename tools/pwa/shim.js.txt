/* Gantt PWA shim (2026-11-21, docs/NOTES.md H). Injected first in the hosted page by tools/build-pwa.mjs.
   - google.script.run → fetch to the "Gantt API" router (POST text/plain, so no CORS preflight), with the app key.
   - Main page: the last data seen on this device opens at once (if it's for today), then fwSync(true) catches up.
   - Offline: read-only + banner. Service worker updates: "↻ Update ready" button, never an auto-reload.
   - Database step (2026-11-22, NOTES H3): with a Firebase config, Keep In Mind reads/writes Firestore directly
     (Google sign-in, his account only), works offline (Firestore queues the edits) and syncs live. */
(function () {
  'use strict';
  var API = '__API__', PAGE = '__PAGE__', BUILD = '__BUILD__', FB_CFG = __FIREBASE__;
  var KEY_LS = 'pwa.key', DATA_LS = 'pwa.initial.v1';
  var ls = {
    get: function (k) { try { return localStorage.getItem(k); } catch (e) { return null; } },
    set: function (k, v) { try { localStorage.setItem(k, v); return true; } catch (e) { return false; } },
    del: function (k) { try { localStorage.removeItem(k); } catch (e) {} }
  };
  window.__PWA__ = { build: BUILD, page: PAGE };

  // Key from the setup link (#k=…): the fragment never reaches a server; save it, then strip it from the URL.
  var m = location.hash.match(/[#&]k=([^&]+)/);
  if (m) ls.set(KEY_LS, decodeURIComponent(m[1]));
  // Look + layout settings copied from the old app (setup link &s=…): applied before the app's scripts read them.
  var sm = location.hash.match(/[#&]s=([^&]+)/);
  if (sm) {
    try {
      var st = JSON.parse(decodeURIComponent(sm[1])), n = 0;
      Object.keys(st).forEach(function (k) {
        if (/^(fw\.|focusWorkspace|tt\.ui\.)/.test(k) && typeof st[k] === 'string' && ls.set(k, st[k])) n++;
      });
      window.__PWA__.settingsCopied = n;
    } catch (e) {}
  }
  if (m || sm) { try { history.replaceState(null, '', location.pathname + location.search); } catch (e) {} }
  function key() { return ls.get(KEY_LS) || ''; }

  /* ---------- offline / read-only ---------- */
  var offline = !navigator.onLine;
  function setOffline(on) {
    on = !!on;
    if (on === offline && document.body && document.body.classList.contains('pwa-offline') === on) return;
    offline = on;
    if (!document.body) return;
    document.body.classList.toggle('pwa-offline', on);
    var bar = document.getElementById('pwaOffline');
    if (on && !bar) {
      bar = document.createElement('div');
      bar.id = 'pwaOffline';
      bar.textContent = fb.canEdit() ? 'Offline · only Keep In Mind saves' : 'Offline · read-only';
      document.body.appendChild(bar);
    } else if (!on && bar) bar.remove();
  }
  window.addEventListener('online', function () { setOffline(false); if (typeof window.fwSync === 'function') try { window.fwSync(true); } catch (e) {} });
  window.addEventListener('offline', function () { setOffline(true); });
  document.addEventListener('DOMContentLoaded', function () { if (offline) { offline = !offline; setOffline(true); } });
  function block(e) {
    if (!offline) return;
    if (e.target && e.target.closest && e.target.closest('#pwaKeyBox')) return;
    if (fb.canEdit() && e.target && e.target.closest && e.target.closest(fb.EDIT_SEL)) return;
    e.preventDefault(); e.stopImmediatePropagation();
    flash();
  }
  ['beforeinput', 'paste', 'drop', 'cut'].forEach(function (t) { document.addEventListener(t, block, true); });
  document.addEventListener('click', function (e) {
    if (offline && e.target && e.target.matches && e.target.matches('input[type=checkbox]')) block(e);
  }, true);
  var flashT = 0;
  function flash() {
    var bar = document.getElementById('pwaOffline'); if (!bar) return;
    bar.classList.add('pwa-flash'); clearTimeout(flashT);
    flashT = setTimeout(function () { bar.classList.remove('pwa-flash'); }, 600);
  }

  /* ---------- key prompt ---------- */
  function needKey(msg) {
    if (document.getElementById('pwaKeyBox')) return;
    var show = function () {
      var box = document.createElement('div');
      box.id = 'pwaKeyBox';
      box.innerHTML = '<div><b>App key needed</b><p>' + (msg || 'This device has no key yet.') +
        ' Easiest: open your private /dev link with <code>?app=pwa</code> at the end, then tap “Open the Gantt app”.</p>' +
        '<input id="pwaKeyIn" type="password" autocomplete="off" placeholder="…or paste the key"><button id="pwaKeyOk">Save</button></div>';
      document.body.appendChild(box);
      document.getElementById('pwaKeyOk').onclick = function () {
        var v = document.getElementById('pwaKeyIn').value.trim();
        if (v) { ls.set(KEY_LS, v); location.reload(); }
      };
    };
    if (document.body) show(); else document.addEventListener('DOMContentLoaded', show);
  }
  if (!key()) needKey();

  /* ---------- Firestore: Keep In Mind (database step, 2026-11-22, docs/NOTES.md → H3) ----------
     getKim / saveKim go to Firestore doc gantt/kim = { html, at, base, dev } instead of the server. The SDK loads
     after the page (no cost to the first paint) from this site (cached by the service worker, so it works offline).
     Offline edits are queued by Firestore and sent when back online. Newest edit wins: the rules refuse an update
     whose `at` isn't newer; the refused copy goes to ganttBackup (→ Backup tab via the Sheet mirror).
     Not signed in, no doc yet, or the SDK not ready within 4 s → the old server path (getKim / saveKim). */
  var fb = (function () {
    var off = { handles: function () { return false; }, canEdit: function () { return false; }, EDIT_SEL: '#kimScroll' };
    if (!FB_CFG || !FB_CFG.apiKey || PAGE !== 'index') return off;
    var NAMES = { getKim: 1, saveKim: 1 }, UNCONF = 'pwa.kim.unconfirmed', DEV_LS = 'pwa.device';
    var OWNER = String(FB_CFG.owner || '').toLowerCase(), cfg = Object.assign({}, FB_CFG);
    delete cfg.owner; delete cfg._note;
    var db = null, user = null, lastAt = null, watching = false;
    var dev = ls.get(DEV_LS);
    if (!dev) { dev = (/Android|iPhone|iPad/i.test(navigator.userAgent) ? 'phone-' : 'pc-') + Math.random().toString(36).slice(2, 8); ls.set(DEV_LS, dev); }

    function load(src) {
      return new Promise(function (ok, bad) {
        var el = document.createElement('script'); el.src = src; el.async = false;
        el.onload = ok; el.onerror = function () { bad(new Error('Couldn’t load ' + src)); };
        document.head.appendChild(el);
      });
    }
    var boot = new Promise(function (ok) {
      if (document.readyState === 'complete') setTimeout(ok, 200);
      else window.addEventListener('load', function () { setTimeout(ok, 200); });
    }).then(function () {
      return Promise.all(['fb/firebase-app-compat.js', 'fb/firebase-auth-compat.js', 'fb/firebase-firestore-compat.js'].map(load));
    }).then(function () {
      firebase.initializeApp(cfg);
      db = firebase.firestore();
      return db.enablePersistence({ synchronizeTabs: true }).catch(function (e) { console.warn('Firestore offline copy off: ' + (e.code || e)); });
    }).then(function () {
      return new Promise(function (ok) {
        firebase.auth().onAuthStateChanged(function (u) {
          if (u && String(u.email || '').toLowerCase() !== OWNER) { firebase.auth().signOut(); u = null; }
          user = u || null;
          signButton(!user);
          if (user) watch();
          ok();
        });
      });
    });
    boot.catch(function (e) { console.warn('Firebase: ' + (e && e.message || e)); });

    function withTimeout(p, ms) {
      return Promise.race([p, new Promise(function (_, bad) { setTimeout(function () { bad(new Error('timeout')); }, ms); })]);
    }
    function doc() { return db.collection('gantt').doc('kim'); }
    function shape(snap) {
      var d = snap && snap.exists ? snap.data() : null;
      return d ? { html: String(d.html || ''), h: 'f' + d.at, at: Number(d.at) || 0, fresh: false, build: '' } : null;
    }
    function read() {
      var cacheOnly = function () { return doc().get({ source: 'cache' }).catch(function () { return null; }); };
      return (navigator.onLine ? doc().get({ source: 'server' }).catch(cacheOnly) : cacheOnly()).then(shape);
    }
    function backup(html, reason) {
      db.collection('ganttBackup').add({ area: 'kim', html: String(html || ''), at: Date.now(), reason: reason, dev: dev })
        .catch(function (e) { console.warn('Backup not saved: ' + (e.code || e)); });
    }
    function unconf() { try { return JSON.parse(ls.get(UNCONF) || 'null'); } catch (e) { return null; } }
    function save(html, baseH) {
      return read().then(function (cur) {
        if (cur && cur.h !== baseH) return { conflict: true, html: cur.html, h: cur.h, build: '' };
        html = String(html || '');
        if (cur && cur.html.length > 400 && html.length < cur.html.length * 0.6) backup(cur.html, 'big shrink');
        var at = Math.max(Date.now(), (cur ? cur.at : 0) + 1);
        var rec = { html: html, at: at, base: cur ? cur.at : 0, dev: dev };
        ls.set(UNCONF, JSON.stringify({ html: html, at: at }));
        // Not awaited: offline, this only resolves once the queued edit reaches the server.
        doc().set(rec).then(function () {
          var u = unconf(); if (u && u.at === at) ls.del(UNCONF);
        }, function (e) {
          if (e && e.code === 'permission-denied') {
            backup(html, 'older edit lost (newest wins)');
            var u = unconf(); if (u && u.at === at) ls.del(UNCONF);
            if (typeof window.fwKimReload === 'function') window.fwKimReload();
          } else console.warn('Keep In Mind not saved to the database: ' + (e.code || e));
        });
        lastAt = at;
        return { h: 'f' + at, at: at, build: '' };
      });
    }
    // Live sync: another device saved → the list merges it in. Also settles an edit left over from an earlier
    // session: confirmed → forget it; the server kept a different copy → the leftover goes to Backup.
    function watch() {
      if (watching) return; watching = true;
      doc().onSnapshot({ includeMetadataChanges: true }, function (snap) {
        if (snap.metadata.fromCache || snap.metadata.hasPendingWrites || !snap.exists) return;
        var at = Number(snap.data().at) || 0, u = unconf();
        if (u) { if (u.at !== at) backup(u.html, 'older edit lost (newest wins)'); ls.del(UNCONF); }
        if (lastAt !== null && at !== lastAt && typeof window.fwKimReload === 'function') window.fwKimReload();
        lastAt = at;
      }, function (e) { watching = false; console.warn('Keep In Mind live sync stopped: ' + (e.code || e)); });
    }
    function signButton(show) {
      if (!document.body) { document.addEventListener('DOMContentLoaded', function () { signButton(show); }); return; }
      var b = document.getElementById('pwaSignIn');
      if (show && !b) {
        b = document.createElement('button');
        b.id = 'pwaSignIn'; b.type = 'button'; b.textContent = 'Sign in · Keep In Mind sync';
        b.onclick = signIn;
        document.body.appendChild(b);
      } else if (!show && b) b.remove();
    }
    function signIn() {
      var p = new firebase.auth.GoogleAuthProvider();
      p.setCustomParameters({ login_hint: OWNER, prompt: 'select_account' });
      firebase.auth().signInWithPopup(p).catch(function (e) {
        if (e && /popup-blocked|operation-not-supported/.test(e.code || '')) return firebase.auth().signInWithRedirect(p);
        if (e && e.code !== 'auth/popup-closed-by-user' && e.code !== 'auth/cancelled-popup-request') alert('Sign-in didn’t work: ' + (e.message || e));
      });
    }
    return {
      EDIT_SEL: '#kimScroll',
      handles: function (name) { return !!NAMES[name]; },
      canEdit: function () { return !!user; },
      run: function (name, args) {
        return withTimeout(boot, 4000).then(function () {
          if (!user) return null;
          return name === 'getKim' ? read() : save(args[0], args[1]);
        }).catch(function (e) { console.warn('Firestore → server: ' + (e && e.message || e)); return null; })
          .then(function (r) { return r || call(name, args); });
      }
    };
  })();

  /* ---------- google.script.run → fetch ---------- */
  function call(fn, args) {
    return fetch(API, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ k: key(), fn: fn, a: args })
    }).then(function (res) {
      if (!res.ok) throw new Error('Server error ' + res.status);
      return res.json();
    }, function () {
      setOffline(true);
      throw new Error('Offline (can’t reach the server)');
    }).then(function (j) {
      if (offline) setOffline(false);
      if (!j || !j.ok) {
        if (j && j.auth) { ls.del(KEY_LS); needKey('That key didn’t work.'); }
        throw new Error((j && j.e) || 'Server error');
      }
      return j.r;
    });
  }
  function runner(ok, bad, user) {
    return new Proxy({}, {
      get: function (_, name) {
        if (name === 'withSuccessHandler') return function (f) { return runner(f, bad, user); };
        if (name === 'withFailureHandler') return function (f) { return runner(ok, f, user); };
        if (name === 'withUserObject') return function (u) { return runner(ok, bad, u); };
        if (typeof name !== 'string') return undefined;
        return function () {
          var args = Array.prototype.slice.call(arguments);
          (fb.handles(name) ? fb.run(name, args) : call(name, args)).then(function (r) {
            if (name === 'getWorkspaceData' && PAGE === 'index') saveData(r);
            if (ok) ok(r, user);
          }, function (err) {
            if (bad) bad(err, user); else console.error(name + ': ' + err.message);
          });
        };
      }
    });
  }
  window.google = window.google || {};
  window.google.script = {
    run: runner(null, null, undefined),
    host: { close: function () {}, setHeight: function () {}, setWidth: function () {}, editor: { focus: function () {} }, origin: location.origin },
    url: { getLocation: function (cb) { cb({ hash: location.hash.slice(1), parameter: {}, parameters: {} }); } }
  };

  /* ---------- main page: open from the last data on this device ---------- */
  function today() {
    var d = new Date();
    return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
  }
  function saveData(r) {
    if (!r || !Array.isArray(r.days) || !r.today) return;
    try {
      var copy = Object.assign({}, r); delete copy.timing;
      if (!ls.set(DATA_LS, JSON.stringify({ at: Date.now(), data: copy }))) ls.del(DATA_LS);
    } catch (e) {}
  }
  window.__PWA_INITIAL__ = null;
  if (PAGE === 'index') {
    var cached = null;
    try { cached = JSON.parse(ls.get(DATA_LS) || 'null'); } catch (e) {}
    if (cached && cached.data && (cached.data.today === today() || offline)) {
      window.__PWA_INITIAL__ = cached.data;
      window.__PWA__.fromCache = cached.at;
      window.addEventListener('load', function () {
        // Catch up right away (days, Reference, imports); the stored copy refreshes a bit later.
        setTimeout(function () { if (!offline && typeof window.fwSync === 'function') try { window.fwSync(true); } catch (e) {} }, 50);
        setTimeout(refreshCopy, 20000);
      });
    } else {
      window.addEventListener('load', function () { setTimeout(refreshCopy, 20000); });
    }
  }
  function refreshCopy() {
    if (offline || document.hidden) return;
    call('getWorkspaceData', [{ fast: true }]).then(saveData, function () {});
  }

  /* ---------- service worker + update button ---------- */
  if ('serviceWorker' in navigator) {
    // Reload only when "↻ Update ready" was tapped; the first install also changes the controller (clients.claim),
    // and reloading then made the page jump on open (2026-11-21).
    var wantReload = false, reloading = false;
    navigator.serviceWorker.addEventListener('controllerchange', function () {
      if (!wantReload || reloading) return; reloading = true; location.reload();
    });
    window.addEventListener('load', function () {
      navigator.serviceWorker.register('sw.js').then(function (reg) {
        function offer(w) {
          if (!w || !navigator.serviceWorker.controller || document.getElementById('pwaUpdate')) return;
          var b = document.createElement('button');
          b.id = 'pwaUpdate'; b.type = 'button'; b.textContent = '↻ Update ready';
          b.onclick = function () {
            b.textContent = 'Saving…'; b.disabled = true;
            try { if (typeof window.fwFlushAll === 'function') window.fwFlushAll(); } catch (e) {}
            setTimeout(function () { wantReload = true; w.postMessage('skipWaiting'); }, 1200);
          };
          document.body.appendChild(b);
        }
        if (reg.waiting) offer(reg.waiting);
        reg.addEventListener('updatefound', function () {
          var w = reg.installing;
          if (w) w.addEventListener('statechange', function () { if (w.state === 'installed') offer(w); });
        });
        var check = function () { if (!document.hidden && navigator.onLine) reg.update().catch(function () {}); };
        document.addEventListener('visibilitychange', check);
        setInterval(check, 10 * 60 * 1000);
      }).catch(function (e) { console.warn('SW', e); });
    });
  }
})();
